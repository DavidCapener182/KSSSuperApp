import assert from'node:assert/strict';import{readFile}from'node:fs/promises';import{createRequire}from'node:module';import{test}from'node:test';import{fileURLToPath}from'node:url';const require=createRequire(fileURLToPath(new URL('../../../kss-ui-shared/package.json',import.meta.url))),postcss=require('postcss'),source=process.env.KSS_CUSTODY_CSS??fileURLToPath(new URL('../../../kss-ui-shared/src/components/assets-workspace.module.css',import.meta.url));
test('Assets family Sheet retains width contract and adds desktop/phone inner insets',async()=>{const ast=postcss.parse(await readFile(source,'utf8'));const rules=[];ast.walkRules('.sheet',rule=>rules.push(rule));const desktop=rules.find(rule=>rule.parent.type==='root'),phone=rules.find(rule=>rule.parent.type==='atrule'&&/max-width:\s*700px/.test(rule.parent.params)),get=(rule,prop)=>rule.nodes.find(node=>node.prop===prop)?.value;assert.equal(get(desktop,'width'),'min(100%,760px)');assert.equal(get(desktop,'max-width'),'760px');assert.equal(get(desktop,'padding'),'24px');assert.equal(get(phone,'max-width'),'100%');assert.equal(get(phone,'padding'),'16px');});
test('Assets Sheet title reserves trailing space for its existing Close control',async()=>{const ast=postcss.parse(await readFile(source,'utf8'));let value;ast.walkRules(rule=>{if(rule.selector==='.sheet > :global([data-slot="sheet-title"])')value=rule.nodes.find(node=>node.prop==='padding-inline-end')?.value;});assert.equal(value,'32px');});
