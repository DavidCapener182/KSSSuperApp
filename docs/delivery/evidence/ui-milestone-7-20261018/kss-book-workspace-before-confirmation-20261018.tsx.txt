type Service = {id:string;name:string;siteName:string};
type History = {kind?:string;version?:number;revision?:number;note?:string;body?:string;reason?:string;actor:string;recordedAt:string};
type Item = {id:string;status:string;revision:number;entryId:string;body:string;occurredAt:string;recordedAt:string;author:string;authorId:string;history:History[]};
type Note = {id:string;type:string;source:string;version:number;body:string;occurredAt:string;recordedAt:string;author:string;authorId:string;itemId?:string;history:History[]};
type Handover = {id:string;revision:number;from:string;until:string;createdAt:string;createdBy:string;closedAt?:string;closeReason?:string;events:History[];acknowledgements:{revision:number;receiver:string;receiverId:string;recordedAt:string}[];acknowledgedByYouAt?:string};
type Book = {service:Service;canWrite:boolean;manager:boolean;actorId:string;openTotal:number;openItems:Item[];recentNotes:Note[];handover:Handover|null};
const types:[string,string][] = [["ROUTINE_OBSERVATION","Routine observation"],["VISITOR_CONTRACTOR_DELIVERY","Contractor, visitor or delivery"],["KEYS_EQUIPMENT","Keys or equipment"],["MAINTENANCE","Maintenance"],["CLIENT_INSTRUCTION","Client instruction"],["OUTSTANDING_ITEM","Outstanding item"]];
const label=(v:string)=>types.find(x=>x[0]===v)?.[1]??v.replaceAll("_"," ").toLowerCase();
const fmt=(v:string)=>new Date(v).toLocaleString("en-GB",{timeZone:"Europe/London",dateStyle:"medium",timeStyle:"short"});
const local=(v:Date|string)=>{const p=Object.fromEntries(new Intl.DateTimeFormat("en-GB",{timeZone:"Europe/London",year:"numeric",month:"2-digit",day:"2-digit",hour:"2-digit",minute:"2-digit",hourCycle:"h23"}).formatToParts(new Date(v)).map(x=>[x.type,x.value]));return `${p.year}-${p.month}-${p.day}T${p.hour}:${p.minute}`};
function candidates(value:string){const [date,time]=value.split("T");if(!date||!time)return [];const [y,m,d]=date.split("-").map(Number),[h,min]=time.split(":").map(Number);const target=Date.UTC(y,m-1,d,h,min);const found:{instant:string;offset:string}[]=[];for(let delta=-120;delta<=120;delta+=15){const instant=new Date(target+delta*60000);if(local(instant)===value){const sign=delta<=0?"+":"-";const abs=Math.abs(delta);found.push({instant:instant.toISOString(),offset:`${sign}${String(Math.floor(abs/60)).padStart(2,"0")}:${String(abs%60).padStart(2,"0")}`})}}return found;}
function EventTime({occurred,setOccurred,choice,setChoice}:{occurred:string;setOccurred:(v:string)=>void;choice:number;setChoice:(v:number)=>void}){const options=candidates(occurred);return <><label>When it happened · UK local time<input required type="datetime-local" value={occurred} onChange={e=>{setOccurred(e.target.value);setChoice(-1)}} /></label>{options.length>1&&<label>Clock change: which occurrence?<select required value={choice} onChange={e=>setChoice(Number(e.target.value))}><option value={-1}>Choose earlier or later</option><option value={0}>Earlier occurrence</option><option value={1}>Later occurrence</option></select></label>}</>}
export function SiteBookWorkspace({serviceId}:{serviceId?:string}){return <SiteBookScope key={serviceId??"directory"} serviceId={serviceId}/>;}
function SiteBookScope({serviceId}:{serviceId?:string}){
 const active=useRef(true),readSequence=useRef(0);
 const [loading,setLoading]=useState(true);
 const expireView=useCallback(()=>{active.current=false;readSequence.current++},[]);
 useEffect(()=>{active.current=true;return expireView},[expireView]);

 const [handoverBody,setHandoverBody]=useState(""),[handoverFrom,setHandoverFrom]=useState(""),[handoverUntil,setHandoverUntil]=useState("");
 const [itemId,setItemId]=useState(""),[itemAction,setItemAction]=useState("UPDATED"),[itemNote,setItemNote]=useState(""),[itemReason,setItemReason]=useState("");

 const [closeReason,setCloseReason]=useState("");
 const [correctId,setCorrectId]=useState(""),[correctBody,setCorrectBody]=useState(""),[correctReason,setCorrectReason]=useState("");
 const pending=useRef<{signature:string;key:string}|null>(null);

 useEffect(()=>{let live=true;void Promise.resolve().then(reload).catch(e=>{if(live)setError(e.message)});void Promise.resolve().then(()=>{if(live){const now=new Date();setOccurred(local(now));setHandoverFrom(local(new Date(now.getTime()-12*3600000)));setHandoverUntil(local(now));}});return()=>{live=false}},[reload]);

Check the book before acting again.":e instanceof TypeError || !navigator.onLine?"Receipt not confirmed. Check the book before retrying when online; your draft remains on this page.":`${e instanceof Error?e.message:"Not confirmed"} Your draft remains on this page. Check the book before retrying.`);return false;}finally{if(active.current)setBusy(false)}}
 async function findEntries(e:React.FormEvent){e.preventDefault();if(!serviceId||loading)return;const ticket=readSequence.current;setError("");try{const q=new URLSearchParams({mode:"search",search:searchText,type:searchType});const r=await fetch(`/api/site-book/${serviceId}?${q}`,{cache:"no-store"});const j=await r.json();if(!active.current||ticket!==readSequence.current)return;if(!r.ok)throw new Error(j.error??"Search unavailable");setSearchResult(j.search)}catch(e){if(active.current&&ticket===readSequence.current)setError(e instanceof Error?e.message:"Search unavailable")}}

Choose another time.");if(c.length>1&&choice<0)throw new Error("Choose which occurrence of the repeated UK clock time you mean.");return {occurredAt:c[Math.max(0,choice)].instant,local:value,offset:c[Math.max(0,choice)].offset}};

 const locked=busy||loading;
 if(!serviceId)return <section className="site-book-card site-book-directory"><div className="site-book-section-head"><div><p className="site-book-kicker">Authorised books</p><h2>Choose a Site Service</h2><p className="site-book-muted">Each book belongs to one exact Service. Open it to read its current handover and facts.</p></div></div>{error&&<p role="alert" className="site-book-alert">{error}</p>}{services.length?<ul className="site-book-list">{services.map(s=><li key={s.id}><Link href={`/site-book/${s.id}`}><span><strong>{s.name}</strong><small>{s.siteName}</small></span><span className="site-book-open">Open Site Book →</span></Link></li>)}</ul>:<p className="site-book-empty">{loading?"Loading authorised Site Services…":error?"Authorised books are unavailable.":"No Site Service books are currently available to you."}</p>}</section>;
 return <><Link className="crm-record-back" href="/site-book">← All Site Books</Link><header className="site-book-record-header"><h1>{book?.service.name??"Site Book"}</h1><button data-site-book-refresh type="button" className="site-book-secondary" disabled={locked} onClick={()=>void reload().catch(e=>setError(e.message))}>Refresh authorised book</button>{book&&<><div className="crm-record-identity"><span>Site Book</span><strong>Routine operational record</strong></div><p>{book.service.siteName} · {book.openTotal} open items</p></>}</header>{book&&<><nav className="site-book-record-sections" aria-label="Site Book sections"><a href="#site-book-handover">Handover</a><a href="#site-book-open-items">Outstanding</a><a href="#site-book-notes">Recent notes</a><a href="#site-book-history">History</a><a href="#site-book-add">Add a fact</a></nav><RecordSectionTracker label="Site Book sections" /></>}{error&&<p className="site-book-alert" role="alert">{error}</p>}{notice&&<p className="site-book-status" role="status">{notice}</p>}{loading&&book&&<p className="site-book-warning" role="status">Refreshing this book. Displayed facts are from the previous read; actions are disabled.</p>}{!book?<section className="site-book-card"><p className="site-book-muted">{loading?"Loading authorised Site Book…":"The current authorised Site Book is unavailable."}</p></section>:<div className="site-book-grid" aria-busy={loading}><div className="site-book-stack">
 <section id="site-book-handover" className="site-book-card"><p className="site-book-kicker">Start of shift</p><h2>Handover</h2>{book.handover?<><p className="site-book-identity">Revision {book.handover.revision} · {book.handover.closedAt?"Closed":"Current handover"}</p><p><strong>Previous shift:</strong> {fmt(book.handover.from)} – {fmt(book.handover.until)}</p><p className="site-book-muted">From {book.handover.createdBy} · Recorded {fmt(book.handover.createdAt)}</p><ul className="site-book-list">{book.handover.events.map(ev=><li key={ev.revision}><strong>{ev.actor} · {fmt(ev.recordedAt)}</strong><p>{ev.body}</p></li>)}</ul><p className={book.handover.acknowledgedByYouAt?"site-book-status":"site-book-warning"}>{book.handover.acknowledgedByYouAt?`Acknowledged by you · ${fmt(book.handover.acknowledgedByYouAt)}`:book.handover.closedAt?"Closed by manager":"Awaiting your acknowledgement"}</p>{book.handover.acknowledgements.length>0&&<details className="site-book-details"><summary>Individual acknowledgements ({book.handover.acknowledgements.filter(a=>a.revision===book.handover?.revision).length})</summary><ul className="site-book-history">{book.handover.acknowledgements.map((a,i)=><li key={i}>{a.receiver} · revision {a.revision} · {fmt(a.recordedAt)}</li>)}</ul></details>}{book.canWrite&&!book.manager&&!book.handover.closedAt&&!book.handover.acknowledgedByYouAt&&<button className="site-book-primary" disabled={locked} onClick={()=>act({action:"handoverAcknowledge",handoverId:book.handover!.id,revision:book.handover!.revision},"Your acknowledgement was recorded.")}>Acknowledge this handover</button>}{book.canWrite&&!book.handover.closedAt&&<SiteBookPanel label="Contribute to handover" title="Add handover contribution" context={`Current handover revision ${book.handover.revision} · ${book.service.name}.
