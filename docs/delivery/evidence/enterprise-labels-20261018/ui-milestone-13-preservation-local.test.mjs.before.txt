import {test} from 'node:test';import assert from 'node:assert/strict';import{readFileSync,readdirSync}from'node:fs';import{createHash}from'node:crypto';import{resolve}from'node:path';import{fileURLToPath}from'node:url';const root=fileURLToPath(new URL('../',import.meta.url)),e=resolve(root,'docs/delivery/evidence/ui-milestone-13-20261018'),load=n=>JSON.parse(readFileSync(resolve(e,n),'utf8')),hash=p=>createHash('sha256').update(readFileSync(resolve(root,p))).digest('hex');test('milestone13 exactly pins all11 authorised products and426 preserved published source files',()=>{const products=load('product-manifest.json'),unchanged=load('non-owned-source-manifest.json');assert.equal(products.length,11);assert.equal(unchanged.length,426);const expected=new Set();for(const row of [...products,...unchanged]){assert.ok(!expected.has(row.file),row.file);expected.add(row.file);assert.equal(hash(row.file),row.sha256,row.file);}function files(dir){return readdirSync(resolve(root,dir),{withFileTypes:true}).flatMap(d=>d.isDirectory()?files(dir+'/'+d.name):[dir+'/'+d.name]);}assert.deepEqual(files('src').sort(),[...expected].sort());});
